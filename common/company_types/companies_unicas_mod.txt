	icon = "gfx/interface/icons/company_icons/historical_company_icons/argentina_centro_vitivinicola_nacional.dds"

	flavored_company = yes
	
	building_types = {  
		building_wheat_farm
		building_maize_farm
	}

	extension_building_types = {
		building_tobacco_plantation
		building_vineyard_plantation
	}
	
	possible_prestige_goods = {
		prestige_good_generic_grain
	}

	potential = {
		has_interest_marker_in_region = region_la_plata
		OR = {
			c:ARG ?= this
			c:PLT ?= this
		}
	}
	
	attainable = {
		hidden_trigger = { # Hide attainable if it's always fulfilled by possible
			any_scope_state = {
				state_region = s:STATE_BUENOS_AIRES
			}
		}
	}		
	
	possible = {
		any_scope_state = {
			state_region = s:STATE_BUENOS_AIRES
			is_incorporated = yes
			any_scope_building = {
				is_building_type = building_wheat_farm
				level >= 10
			}
		}
	}
	
	prosperity_modifier = {
	
		building_wheat_farm_throughput_add = 0.25
		building_maize_farm_throughput_add = 0.25
		building_group_bg_plantations_throughput_add = 0.1
		state_birth_rate_mult = 0.05
	}	
	
	ai_weight = {
		value = 3 # Higher base value for flavored companies
	}
}
